short-ok CurvedVal.cpp t -- a curve's parameter, t in the curve's own notation as it runs from 0 to 1
allow offset anAlpha255 -- an alpha on the 0 to 255 scale, beside the 0 to 1 alpha it is made from
allow offset aGlowAlpha255 -- the glow's alpha on the 0 to 255 scale
allow offset theAllow1200 -- the 1200-pixel art size, the "ArtRes" value it allows
allow offset mAllowArtRes1200 -- the 1200-pixel art size, the "ArtRes" value it allows
